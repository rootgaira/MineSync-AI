## 1. Project Overview

MineSync AI is a web-based intelligent document processing platform for mining and geological data.

**Core Purpose:**
- Accept mining/geological documents (PDFs, images, Excel, Word)
- Extract structured data using AI
- Validate extracted information
- Enable AI-powered querying with source traceability
- Generate official parliamentary responses
- Create analytics and reports

**Architecture Type:** Three-tier (Frontend-Backend-Database)

**Technology Stack:**
- Frontend: React 18+ (JavaScript)
- Backend: FastAPI (Python)
- Database: PostgreSQL (SQL)
- Communication: HTTP/REST with JSON

### Key Files Explained

#### main.py - FastAPI Application

```python
# FastAPI is a Python web framework for building REST APIs
# It automatically:
# 1. Validates requests using Pydantic schemas
# 2. Converts responses to JSON
# 3. Generates API documentation (/docs)
# 4. Handles errors gracefully

# Key Components:

1. STARTUP EVENT
# @app.on_event("startup")
# When backend starts:
# - Calls init_db() to create tables if they don't exist
# - Prints success message

2. CORS MIDDLEWARE
# Allows frontend (localhost:3000) to call backend (localhost:8000)
# Without this, browser blocks cross-origin requests

3. ENDPOINTS
# Each endpoint:
# - Accepts HTTP request with specific method (GET, POST, etc.)
# - Validates input using Pydantic schema
# - Interacts with database
# - Returns response (automatically converted to JSON)

# Example: POST /documents/upload
@app.post("/documents/upload")
async def upload_document(file: UploadFile = File(...),
                         document_type: str = Query(...),
                         db: Session = Depends(get_db)):
    """
    Flow:
    1. Frontend sends file + document_type
    2. Validate file extension (must be .pdf, .docx, etc.)
    3. Save file to disk (backend/uploads/)
    4. Get file size
    5. Create Document record in database
    6. Return new document as JSON
    """

    # Validate file type
    allowed_extensions = {".pdf", ".docx", ".xlsx", ".png", ".jpg", ".jpeg"}
    file_ext = Path(file.filename).suffix.lower()
    if file_ext not in allowed_extensions:
        # Return HTTP 400 error if invalid
        raise HTTPException(status_code=400, detail="File type not supported")

    # Save file to disk
    file_path = UPLOAD_DIR / file.filename
    with open(file_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)  # Copy uploaded file to disk

    # Create database record
    db_document = Document(
        filename=file.filename,
        file_path=str(file_path),
        document_type=document_type,
        status=DocumentStatus.UPLOADED,
        file_size=file_path.stat().st_size,
        metadata={"uploaded_by": "system", "original_name": file.filename}
    )
    db.add(db_document)          # Add to session
    db.commit()                   # Save to database
    db.refresh(db_document)       # Reload to get auto-generated ID

    # Return as JSON
    return DocumentResponse.from_orm(db_document)
```

#### models.py - Database Models

```python
# SQLAlchemy ORM maps Python classes to database tables
# Each class = One database table

# Example: Document model

class Document(Base):
    __tablename__ = "documents"  # Table name in PostgreSQL

    # Columns (each becomes a database column)
    id = Column(Integer, primary_key=True, index=True)
    filename = Column(String, index=True)
    file_path = Column(String)
    document_type = Column(String)
    status = Column(SQLEnum(DocumentStatus))  # Enum: uploaded, processing, processed, failed
    upload_date = Column(DateTime, default=datetime.utcnow)
    file_size = Column(Integer)
    metadata = Column(JSON, nullable=True)  # Flexible JSON data

    # Relationship (connects to DocumentPage table)
    pages = relationship("DocumentPage", back_populates="document")

# When you create a Document:
doc = Document(filename="report.pdf", document_type="production_report", ...)
db.add(doc)
db.commit()  # INSERT INTO documents (...) VALUES (...)

# 11 Models (Tables):
1. Document        - Uploaded files
2. DocumentPage    - Individual pages/chunks
3. ExtractedRecord - Structured data extracted
4. ValidationResult - Validation checks
5. Query           - User questions
6. QuerySource     - Document references for queries
7. Report          - Generated reports
8. Topic           - Keyword analysis
9. User            - User management (Phase 8)
10. AuditLog       - Activity logging
```

#### schemas.py - Request/Response Validation

```python
# Pydantic schemas validate incoming requests and format responses

# Example:

class DocumentCreate(BaseModel):
    filename: str          # Must be string
    document_type: str     # Must be string

    # When frontend POSTs:
    # {"filename": "report.pdf", "document_type": "production_report"}
    # Pydantic validates types and returns DocumentCreate object

class DocumentResponse(BaseModel):
    id: int
    filename: str
    document_type: str
    status: str
    upload_date: datetime
    file_size: int
    metadata: Optional[Dict[str, Any]] = None

    class Config:
        from_attributes = True  # Convert ORM to dict

    # When returning document, FastAPI uses this schema to format JSON response:
    # {"id": 1, "filename": "report.pdf", ...}

# Benefits:
1. Type validation - catches errors early
2. Auto docs - /docs shows what each endpoint accepts
3. Auto JSON conversion - no manual serialization needed
4. Clear contracts between frontend and backend
```

#### database.py - Database Connection

```python
# Manages PostgreSQL connection

# Load .env file
load_dotenv()
DATABASE_URL = os.getenv("DATABASE_URL")
# Example: "postgresql://postgres:postgres@localhost:5432/minesync_ai"

# Create engine (connection pool)
engine = create_engine(DATABASE_URL)

# Create session factory
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# Dependency function for endpoints
def get_db():
    db = SessionLocal()  # Create new session
    try:
        yield db         # Give to endpoint
    finally:
        db.close()       # Always close after use

# Usage in endpoint:
@app.get("/documents")
async def list_documents(db: Session = Depends(get_db)):
    # FastAPI automatically calls get_db() and passes result as 'db'
    documents = db.query(Document).all()
    return documents
```

## 4. Database Design

### Schema Overview

```
DOCUMENTS TABLE
├── id (primary key)
├── filename
├── file_path
├── document_type
├── status (uploaded, processing, processed, failed)
├── upload_date
├── file_size
└── metadata (JSON)
    └── Relationships:
        └── pages (many) → DOCUMENT_PAGES
        └── extractions (many) → EXTRACTED_RECORDS

DOCUMENT_PAGES TABLE
├── id (primary key)
├── document_id (foreign key)
├── page_number
├── raw_text (OCR result, Phase 2)
└── extracted_text (cleaned text)

EXTRACTED_RECORDS TABLE
├── id (primary key)
├── source_document_id (foreign key)
├── source_page
├── subsidiary
├── mine
├── financial_year
├── production_value
├── production_unit
├── target_value
├── extracted_data (JSON with full details)
├── validation_status
├── confidence_score
└── extraction_timestamp

VALIDATION_RESULTS TABLE
├── id (primary key)
├── record_id (foreign key)
├── validation_type (unit_normalization, cross_document, anomaly)
├── status (pass, warning, fail)
├── message (description of issue)
├── resolved (boolean - human reviewed?)
└── created_at

QUERIES TABLE
├── id (primary key)
├── user_question
├── query_type (standard, parliamentary)
└── metadata (JSON with filter info)

QUERY_SOURCES TABLE
├── id (primary key)
├── query_id (foreign key)
├── source_document_id (foreign key)
└── relevance_score

REPORTS TABLE
├── id (primary key)
├── title
├── report_type (production, geological, parliamentary)
├── filters (JSON - what was included)
├── content (generated report)
├── generated_at
└── file_path (if exported)

TOPICS TABLE
├── id (primary key)
├── name
├── keyword_frequency (JSON - {"keyword": count, ...})
├── document_count
└── created_at

AUDIT_LOGS TABLE
├── id (primary key)
├── action (what happened)
├── entity_type (what was affected)
├── entity_id (which record)
├── user_id (who did it)
├── timestamp
└── details (JSON - extra info)
```

## 5. API Endpoints

### Document Management

#### POST /documents/upload
```

- Form data with file upload
- Query parameter: document_type

Example:
curl -X POST "http://localhost:8000/documents/upload?document_type=production_report" \
  -F "file=@report.pdf"

Response:
{
  "id": 1,
  "filename": "report.pdf",
  "document_type": "production_report",
  "status": "uploaded",
  "upload_date": "2026-09-08T08:30:00",
  "file_size": 250000

Backend Logic:
1. Validate file extension
2. Save file to backend/uploads/
3. Create Document record in DB
4. Return created document
```

```
Request:
- Query parameters: skip (default 0), limit (default 10)

Example:
```

### SQLAlchemy (Database ORM)

```python
# ORM = Object-Relational Mapping
# Maps database tables to Python classes

from sqlalchemy import Column, String, Integer, DateTime
from sqlalchemy.ext.declarative import declarative_base

Base = declarative_base()

class Document(Base):
    __tablename__ = "documents"
    id = Column(Integer, primary_key=True)
    filename = Column(String)
    status = Column(String)

# Python side:
doc = Document(filename="report.pdf", status="uploaded")
db.add(doc)
db.commit()

# SQL generated automatically:
# INSERT INTO documents (filename, status)
# VALUES ('report.pdf', 'uploaded');

# Benefits:
# - No SQL injection (SQLAlchemy handles escaping)
# - Type safety (Python types)
# - Easier to maintain (no SQL strings scattered in code)
# - Can switch databases (PostgreSQL, MySQL, SQLite)
```

### PostgreSQL (Database)

```sql
-- Relational database stores data in tables
-- Each table has columns and rows

CREATE TABLE documents (
    id SERIAL PRIMARY KEY,              -- Auto-incrementing integer
    filename VARCHAR NOT NULL,           -- Text field (required)
    file_path VARCHAR NOT NULL,
    document_type VARCHAR,
    status VARCHAR DEFAULT 'uploaded',   -- Default value
    upload_date TIMESTAMP DEFAULT NOW(), -- Current time
    file_size INTEGER,
    metadata JSONB                       -- JSON data
);

-- Index on commonly searched columns for speed
CREATE INDEX idx_documents_status ON documents(status);

-- Querying:
SELECT * FROM documents WHERE status = 'processed';
SELECT COUNT(*) FROM documents;
SELECT * FROM documents ORDER BY upload_date DESC LIMIT 10;
```

#### 3. API Endpoint (FastAPI)
```python
@app.get("/documents")  # Route
async def get_documents(  # Handler
    skip: int = Query(0),  # Parameter with default
    limit: int = Query(10),
    db: Session = Depends(get_db)  # Dependency injection
) -> DocumentListResponse:  # Return type
    """Docstring auto-used in /docs"""

    # Business logic
    docs = db.query(Document).offset(skip).limit(limit).all()
    total = db.query(Document).count()

    # Response
    return DocumentListResponse(
        documents=[DocumentResponse.from_orm(doc) for doc in docs],
        total=total
    )
```

```python
# Read all

db.commit()

# Delete
```
